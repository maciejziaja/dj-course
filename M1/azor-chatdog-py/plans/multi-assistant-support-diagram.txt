    UI->>CH: handle_command()
    CH->>CS: load_from_file(session_id)
    CS->>SF: load_session_history(session_id)
    SF->>F: Read JSON file
    F-->>SF: {history, title, assistant_id, ...}
    SF-->>CS: history, title, assistant_id
    CS->>R: get_assistant_by_id(assistant_id or "azor")
    R->>A: create_[assistant]()
    A-->>CS: Assistant instance
    CS->>CS: __init__(assistant, session_id, history, title)
    CS->>CS: _initialize_llm_session()
    CS-->>CH: ChatSession restored
    
    Note over U,F: Scenario 3: Switching Assistant
    U->>UI: /assistant perfectionist
    UI->>CH: handle_command()
    CH->>AS: switch_assistant_command(session, "perfectionist")
    AS->>R: validate assistant_id("perfectionist")
    R-->>AS: Valid
    AS->>CS: switch_assistant("perfectionist")
    CS->>R: get_assistant_by_id("perfectionist")
    R->>A: create_perfectionist_assistant()
    A-->>CS: New Assistant instance
    CS->>CS: Add change message to history
    Note right of CS: [SYSTEM: Change AZOR → PERFEKCJONISTA]
    CS->>CS: self.assistant = new_assistant
    CS->>CS: self.assistant_id = "perfectionist"
    CS->>CS: _initialize_llm_session()
    CS->>SF: save_to_file()
    SF->>F: Write JSON with new assistant_id
    CS-->>AS: Success
    AS-->>CH: Success

    class Assistant {
        +str system_prompt
        +str name
        +get_system_prompt() str
        +get_name() str

    class AssistantRegistry {
        +Dict[str, Callable] _factories
        +get_assistant_by_id(id: str) Assistant
        +list_available_assistants() List[Dict]
        +register_assistant(id: str, factory: Callable) None
    }
    
    class ChatSession {
        +Assistant assistant
        +str assistant_id
        +str session_id
        +str|None title
        +List history
        +switch_assistant(assistant_id: str) None
        +save_to_file() tuple[bool, str|None]
        +load_from_file(session_id: str) tuple[ChatSession|None, str|None]
        -_get_assistant_id_from_assistant() str
    }
    
    class SessionManager {
        +ChatSession _current_session
        +create_new_session(assistant_id: str|None, save_current: bool) tuple
        +switch_to_session(session_id: str) tuple
        +initialize_from_cli(session_id: str|None) ChatSession
        +remove_current_session_and_create_new() tuple
    }
    
    class SessionFiles {
        +save_session_history(id, history, system_prompt, model_name, title, assistant_id) tuple
        +load_session_history(session_id: str) tuple[List, str|None, str|None, str|None]
    }
    
    AssistantRegistry --> Assistant : creates
    ChatSession --> Assistant : contains
    ChatSession --> SessionFiles : uses
    SessionManager --> ChatSession : manages
    SessionManager --> AssistantRegistry : uses
